```python
class LLMRequest(BaseModel):
    system: str; user: str; images: list[bytes] = []; model: str | None = None; temperature: float = 0.2; max_tokens: int = 512
class LLMResult(BaseModel):
    parsed: Any            # instance of the requested schema (BaseModel); model_config arbitrary_types_allowed
    raw_text: str          # scrubbed with scrub_raw()
    model: str; duration_ms: int
class LLMHealth(BaseModel): ok: bool; kind: LLMKind; detail: str = ""; models: list[str] = []
class LLMOutputInvalid(Exception):  raw_text: str (scrubbed); errors: list[str]
class LLMUnavailable(Exception): ...
def scrub_raw(text: str) -> str      # same token patterns as logging_setup (access_token=, Bearer, IGQ/IGAA/EAA 40+, 60+ char runs) -> "[REDACTED]"
@runtime_checkable
class LLMClient(Protocol):
    @property
    def kind(self) -> LLMKind: ...
    async def generate_structured(self, request: LLMRequest, schema: type[BaseModel]) -> LLMResult: ...
    async def health(self) -> LLMHealth: ...
LLM_MEMBERS = frozenset({"kind", "generate_structured", "health"})
```
