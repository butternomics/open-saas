class CommentDTO(BaseModel): id: str; post_id: str; parent_id: str | None = None; author_id: str | None = None; author_username: str | None = None; text: str = ""; created_at: datetime | None = None; reply_count: int = 0
class UsageInfo(BaseModel):  call_count_pct: float = 0.0; total_time_pct: float = 0.0; total_cputime_pct: float = 0.0; business_use_case_raw: dict | None = None
                             @property near_limit -> bool  # any pct >= 75.0
class CommentPage(BaseModel): items: list[CommentDTO]; next_cursor: str | None = None; usage: UsageInfo = UsageInfo()
class ReplyResult(BaseModel): reply_id: str; raw: dict = {}
class AuthStatus(BaseModel): ok: bool; account_id: str | None = None; username: str | None = None; token_expires_at: datetime | None = None; scopes_documented: list[str] = []; error: str | None = None; checked_at: datetime
DOCUMENTED_SCOPES = ["instagram_business_basic", "instagram_business_manage_comments"]
class ConnectorError(Exception): ...
class AuthError(ConnectorError): ...
class RateLimitError(ConnectorError): retry_after_s: float | None
class TransientError(ConnectorError): ...
class PermanentError(ConnectorError): ...
class UncertainSendError(ConnectorError): ...      # only from post_reply
@runtime_checkable
class Connector(Protocol):
    @property
    def kind(self) -> ConnectorKind: ...
    @property
    def is_mock(self) -> bool: ...
    @property
    def display_name(self) -> str: ...
    async def fetch_posts(self, since: datetime | None = None, limit: int = 25) -> list[PostDTO]: ...
    async def fetch_comments(self, post_id: str, cursor: str | None = None) -> CommentPage: ...
    async def fetch_replies(self, comment_id: str, cursor: str | None = None) -> CommentPage: ...
    async def post_reply(self, comment_id: str, text: str, idempotency_key: str) -> ReplyResult: ...
    async def check_auth(self) -> AuthStatus: ...
    async def refresh_credentials(self) -> AuthStatus: ...
CONNECTOR_MEMBERS = frozenset({"kind", "is_mock", "display_name", "fetch_posts", "fetch_comments", "fetch_replies", "post_reply", "check_auth", "refresh_credentials"})

    def __init__(self, **queues: list[Behavior]) ...

class MockConnector:
    def __init__(self, dataset_path: Path | None = None, scenario: MockScenario | None = None, page_size: int = 25) ...
    kind = ConnectorKind.MOCK; is_mock = True; display_name = "MockConnector (MOCK)"
    sent_replies: list[dict]        # {"comment_id", "text", "idempotency_key", "reply_id"}
    calls: list[tuple[str, dict]]   # every operation with its kwargs, for assertions
    # behaviour mapping: "near_limit" -> usage pcts 90.0 on the page; "rate_limit" -> raise RateLimitError(retry_after_s=60); "auth_failure" -> raise AuthError("token expired") (check_auth returns ok=False with error instead of raising);
    # "timeout" -> post_reply raises UncertainSendError, reads raise TransientError; "permanent_error" -> raise PermanentError("mock permanent")
    # post_reply appends a reply CommentDTO authored by MOCK_ACCOUNT_ID to the in-memory thread so fetch_replies sees it; idempotent: same idempotency_key returns the same reply_id without a second append
    # cursor format: "p{page_index}" (opaque to callers)
